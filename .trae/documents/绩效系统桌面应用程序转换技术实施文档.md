# 绩效系统桌面应用程序转换技术实施文档

## 1. 项目概述

### 1.1 转换目标
将基于React + IndexedDB的Web绩效系统转换为Electron桌面应用程序，使用SQLite替代IndexedDB存储，解决浏览器数据不稳定问题。

### 1.2 技术栈对比

**转换前：**
- 前端：React + TypeScript + Vite + Ant Design
- 数据存储：Dexie.js (IndexedDB)
- 后端API：Express.js
- 运行环境：浏览器

**转换后：**
- 前端：React + TypeScript + Ant Design
- 桌面框架：Electron
- 数据存储：SQLite + better-sqlite3
- 后端API：Express.js (集成到主进程)
- 运行环境：桌面应用程序

## 3. 阶段二：SQLite数据库集成

### 3.1 创建数据库管理器

**文件路径：** `electron/database.js`

```javascript
const Database = require('better-sqlite3')
const path = require('path')
const { app } = require('electron')
const fs = require('fs')

class DatabaseManager {
  constructor() {
    const userDataPath = app.getPath('userData')
    const dbPath = path.join(userDataPath, 'performance.db')
    
    // 确保数据目录存在
    if (!fs.existsSync(userDataPath)) {
      fs.mkdirSync(userDataPath, { recursive: true })
    }
    
    this.db = new Database(dbPath)
    this.initTables()
  }

  initTables() {
    // 创建员工表
    this.db.exec(`
      CREATE TABLE IF NOT EXISTS employees (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        empNo TEXT UNIQUE NOT NULL,
        name TEXT NOT NULL,
        position TEXT,
        station TEXT,
        area TEXT,
        workshop TEXT,
        center TEXT,
        company TEXT,
        createdAt DATETIME DEFAULT CURRENT_TIMESTAMP,
        updatedAt DATETIME DEFAULT CURRENT_TIMESTAMP
      )
    `)

    // 创建绩效项目表
    this.db.exec(`
      CREATE TABLE IF NOT EXISTS items (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        paperCode TEXT,
        itemCode TEXT NOT NULL,
        name TEXT NOT NULL,
        category TEXT,
        score REAL,
        description TEXT,
        status TEXT DEFAULT 'active',
        groupName TEXT,
        createdAt DATETIME DEFAULT CURRENT_TIMESTAMP,
        updatedAt DATETIME DEFAULT CURRENT_TIMESTAMP
      )
    `)

    // 创建绩效记录表
    this.db.exec(`
      CREATE TABLE IF NOT EXISTS records (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        empNo TEXT NOT NULL,
        itemCode TEXT NOT NULL,
        month TEXT NOT NULL,
        score REAL,
        remarks TEXT,
        createdAt DATETIME DEFAULT CURRENT_TIMESTAMP,
        updatedAt DATETIME DEFAULT CURRENT_TIMESTAMP,
        FOREIGN KEY (empNo) REFERENCES employees(empNo),
        FOREIGN KEY (itemCode) REFERENCES items(itemCode)
      )
    `)

    // 创建索引
    this.db.exec(`
      CREATE INDEX IF NOT EXISTS idx_employees_empNo ON employees(empNo);
      CREATE INDEX IF NOT EXISTS idx_items_itemCode ON items(itemCode);
      CREATE INDEX IF NOT EXISTS idx_records_empNo ON records(empNo);
      CREATE INDEX IF NOT EXISTS idx_records_itemCode ON records(itemCode);
      CREATE INDEX IF NOT EXISTS idx_records_month ON records(month);
    `)
  }

  // 员工操作
  getEmployees() {
    return this.db.prepare('SELECT * FROM employees ORDER BY createdAt DESC').all()
  }

  addEmployee(employee) {
    const stmt = this.db.prepare(`
      INSERT INTO employees (empNo, name, position, station, area, workshop, center, company)
      VALUES (?, ?, ?, ?, ?, ?, ?, ?)
    `)
    return stmt.run(
      employee.empNo, employee.name, employee.position, employee.station,
      employee.area, employee.workshop, employee.center, employee.company
    )
  }

  updateEmployee(id, employee) {
    const stmt = this.db.prepare(`
      UPDATE employees 
      SET empNo = ?, name = ?, position = ?, station = ?, area = ?, workshop = ?, center = ?, company = ?, updatedAt = CURRENT_TIMESTAMP
      WHERE id = ?
    `)
    return stmt.run(
      employee.empNo, employee.name, employee.position, employee.station,
      employee.area, employee.workshop, employee.center, employee.company, id
    )
  }

  deleteEmployee(id) {
    return this.db.prepare('DELETE FROM employees WHERE id = ?').run(id)
  }

  // 绩效项目操作
  getItems() {
    return this.db.prepare('SELECT * FROM items ORDER BY createdAt DESC').all()
  }

  addItem(item) {
    const stmt = this.db.prepare(`
      INSERT INTO items (paperCode, itemCode, name, category, score, description, status, groupName)
      VALUES (?, ?, ?, ?, ?, ?, ?, ?)
    `)
    return stmt.run(
      item.paperCode, item.itemCode, item.name, item.category,
      item.score, item.description, item.status, item.groupName
    )
  }

  updateItem(id, item) {
    const stmt = this.db.prepare(`
      UPDATE items 
      SET paperCode = ?, itemCode = ?, name = ?, category = ?, score = ?, description = ?, status = ?, groupName = ?, updatedAt = CURRENT_TIMESTAMP
      WHERE id = ?
    `)
    return stmt.run(
      item.paperCode, item.itemCode, item.name, item.category,
      item.score, item.description, item.status, item.groupName, id
    )
  }

  deleteItem(id) {
    return this.db.prepare('DELETE FROM items WHERE id = ?').run(id)
  }

  // 绩效记录操作
  getRecords() {
    return this.db.prepare(`
      SELECT r.*, e.name as employeeName, i.name as itemName
      FROM records r
      LEFT JOIN employees e ON r.empNo = e.empNo
      LEFT JOIN items i ON r.itemCode = i.itemCode
      ORDER BY r.createdAt DESC
    `).all()
  }

  addRecord(record) {
    const stmt = this.db.prepare(`
      INSERT INTO records (empNo, itemCode, month, score, remarks)
      VALUES (?, ?, ?, ?, ?)
    `)
    return stmt.run(record.empNo, record.itemCode, record.month, record.score, record.remarks)
  }

  updateRecord(id, record) {
    const stmt = this.db.prepare(`
      UPDATE records 
      SET empNo = ?, itemCode = ?, month = ?, score = ?, remarks = ?, updatedAt = CURRENT_TIMESTAMP
      WHERE id = ?
    `)
    return stmt.run(record.empNo, record.itemCode, record.month, record.score, record.remarks, id)
  }

  deleteRecord(id) {
    return this.db.prepare('DELETE FROM records WHERE id = ?').run(id)
  }

  // 数据迁移
  migrateFromIndexedDB(data) {
    const transaction = this.db.transaction(() => {
      // 清空现有数据
      this.db.prepare('DELETE FROM records').run()
      this.db.prepare('DELETE FROM items').run()
      this.db.prepare('DELETE FROM employees').run()

      // 迁移员工数据
      if (data.employees) {
        const stmt = this.db.prepare(`
          INSERT INTO employees (empNo, name, position, station, area, workshop, center, company, createdAt, updatedAt)
          VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
        `)
        data.employees.forEach(emp => {
          stmt.run(
            emp.empNo, emp.name, emp.position, emp.station,
            emp.area, emp.workshop, emp.center, emp.company,
            emp.createdAt || new Date().toISOString(),
            emp.updatedAt || new Date().toISOString()
          )
        })
      }

      // 迁移绩效项目数据
      if (data.items) {
        const stmt = this.db.prepare(`
          INSERT INTO items (paperCode, itemCode, name, category, score, description, status, groupName, createdAt, updatedAt)
          VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
        `)
        data.items.forEach(item => {
          stmt.run(
            item.paperCode, item.itemCode, item.name, item.category,
            item.score, item.description, item.status, item.groupName,
            item.createdAt || new Date().toISOString(),
            item.updatedAt || new Date().toISOString()
          )
        })
      }

      // 迁移绩效记录数据
      if (data.records) {
        const stmt = this.db.prepare(`
          INSERT INTO records (empNo, itemCode, month, score, remarks, createdAt, updatedAt)
          VALUES (?, ?, ?, ?, ?, ?, ?)
        `)
        data.records.forEach(record => {
          stmt.run(
            record.empNo, record.itemCode, record.month, record.score, record.remarks,
            record.createdAt || new Date().toISOString(),
            record.updatedAt || new Date().toISOString()
          )
        })
      }
    })

    transaction()
    return { success: true, message: '数据迁移完成' }
  }

  // 数据备份
  backupDatabase() {
    const userDataPath = app.getPath('userData')
    const backupPath = path.join(userDataPath, 'backups')
    
    if (!fs.existsSync(backupPath)) {
      fs.mkdirSync(backupPath, { recursive: true })
    }
    
    const timestamp = new Date().toISOString().replace(/[:.]/g, '-')
    const backupFile = path.join(backupPath, `performance_backup_${timestamp}.db`)
    
    fs.copyFileSync(path.join(userDataPath, 'performance.db'), backupFile)
    return backupFile
  }

  close() {
    this.db.close()
  }
}

module.exports = DatabaseManager
```

## 4. 阶段三：数据迁移方案

### 4.1 创建数据迁移组件

**文件路径：** `src/components/DataMigration.tsx`

```typescript
import React, { useState } from 'react'
import { Button, Card, Steps, Alert, Progress, message } from 'antd'
import { db } from '../database/db'

const { Step } = Steps

interface MigrationData {
  employees: any[]
  items: any[]
  records: any[]
}

const DataMigration: React.FC = () => {
  const [currentStep, setCurrentStep] = useState(0)
  const [migrationData, setMigrationData] = useState<MigrationData | null>(null)
  const [progress, setProgress] = useState(0)
  const [isExporting, setIsExporting] = useState(false)
  const [isMigrating, setIsMigrating] = useState(false)

  const exportFromIndexedDB = async () => {
    setIsExporting(true)
    setProgress(0)
    
    try {
      // 导出员工数据
      setProgress(20)
      const employees = await db.employees.toArray()
      
      // 导出绩效项目数据
      setProgress(50)
      const items = await db.items.toArray()
      
      // 导出绩效记录数据
      setProgress(80)
      const records = await db.records.toArray()
      
      setProgress(100)
      
      const data: MigrationData = {
        employees: employees.map(emp => ({
          ...emp,
          createdAt: emp.createdAt?.toISOString(),
          updatedAt: emp.updatedAt?.toISOString()
        })),
        items: items.map(item => ({
          ...item,
          createdAt: item.createdAt?.toISOString(),
          updatedAt: item.updatedAt?.toISOString()
        })),
        records: records.map(record => ({
          ...record,
          createdAt: record.createdAt?.toISOString(),
          updatedAt: record.updatedAt?.toISOString()
        }))
      }
      
      setMigrationData(data)
      setCurrentStep(1)
      message.success('数据导出成功！')
    } catch (error) {
      message.error('数据导出失败：' + error.message)
    } finally {
      setIsExporting(false)
    }
  }

  const migrateToSQLite = async () => {
    if (!migrationData) return
    
    setIsMigrating(true)
    setProgress(0)
    
    try {
      setProgress(50)
      const result = await window.electronAPI.database.migrateFromIndexedDB(migrationData)
      setProgress(100)
      
      if (result.success) {
        setCurrentStep(2)
        message.success('数据迁移完成！')
      } else {
        message.error('数据迁移失败')
      }
    } catch (error) {
      message.error('数据迁移失败：' + error.message)
    } finally {
      setIsMigrating(false)
    }
  }

  const steps = [
    {
      title: '导出数据',
      description: '从IndexedDB导出现有数据'
    },
    {
      title: '迁移数据',
      description: '将数据导入到SQLite数据库'
    },
    {
      title: '完成',
      description: '数据迁移完成'
    }
  ]

  return (
    <Card title="数据迁移工具" style={{ margin: '20px' }}>
      <Steps current={currentStep} style={{ marginBottom: '20px' }}>
        {steps.map(step => (
          <Step key={step.title} title={step.title} description={step.description} />
        ))}
      </Steps>
      
      {currentStep === 0 && (
        <div>
          <Alert
            message="数据导出"
            description="点击下方按钮从浏览器IndexedDB导出现有数据，这将包括所有员工信息、绩效项目和绩效记录。"
            type="info"
            style={{ marginBottom: '20px' }}
          />
          
          {isExporting && (
            <Progress percent={progress} style={{ marginBottom: '20px' }} />
          )}
          
          <Button 
            type="primary" 
            onClick={exportFromIndexedDB}
            loading={isExporting}
            size="large"
          >
            开始导出数据
          </Button>
        </div>
      )}
      
      {currentStep === 1 && (
        <div>
          <Alert
            message="数据迁移"
            description={`已成功导出 ${migrationData?.employees.length || 0} 名员工、${migrationData?.items.length || 0} 个绩效项目、${migrationData?.records.length || 0} 条绩效记录。点击下方按钮将数据迁移到SQLite数据库。`}
            type="success"
            style={{ marginBottom: '20px' }}
          />
          
          {isMigrating && (
            <Progress percent={progress} style={{ marginBottom: '20px' }} />
          )}
          
          <Button 
            type="primary" 
            onClick={migrateToSQLite}
            loading={isMigrating}
            size="large"
          >
            开始迁移数据
          </Button>
        </div>
      )}
      
      {currentStep === 2 && (
        <div>
          <Alert
            message="迁移完成"
            description="数据迁移已成功完成！现在可以关闭此工具，系统将使用SQLite数据库存储数据。"
            type="success"
          />
        </div>
      )}
    </Card>
  )
}

export default DataMigration
```

### 5.4 创建构建脚本

**文件路径：** `scripts/build.js`

```javascript
const { execSync } = require('child_process')
const fs = require('fs')
const path = require('path')

console.log('开始构建绩效考核系统桌面应用程序...')

try {
  // 1. 清理旧的构建文件
  console.log('1. 清理构建目录...')
  if (fs.existsSync('dist')) {
    fs.rmSync('dist', { recursive: true, force: true })
  }

  // 2. 构建React应用
  console.log('2. 构建React应用...')
  execSync('npm run build', { stdio: 'inherit' })

  // 3. 确保输出目录存在
  console.log('3. 准备打包目录...')
  const outputDir = 'F:/安装包'
  if (!fs.existsSync(outputDir)) {
    fs.mkdirSync(outputDir, { recursive: true })
  }

  // 4. 打包Electron应用
  console.log('4. 打包Electron应用...')
  execSync('electron-builder --publish=never', { stdio: 'inherit' })

  console.log('✅ 构建完成！安装包已生成到 F:/安装包 目录')
} catch (error) {
  console.error('❌ 构建失败:', error.message)
  process.exit(1)
}
```
